def hero_square(src, dst):
    """Cắt ảnh Node lắp ráp thành hình vuông trắng đủ rộng để lọt trong vòng tròn (không cắt mất ăng-ten)."""
    im = Image.open(os.path.join(A, src)).convert("RGB")
    x0, y0, x1, y1 = ImageChops.difference(im, Image.new("RGB", im.size, (255, 255, 255))).point(lambda v: 255 if v > 10 else 0).getbbox()
    cx, cy = (x0 + x1) / 2, (y0 + y1) / 2
    r = max(((x - cx) ** 2 + (y - cy) ** 2) ** 0.5 for x in (x0, x1) for y in (y0, y1)) * 1.06
    side = int(2 * r)
    c = Image.new("RGB", (side, side), (255, 255, 255)); c.paste(im.crop((x0, y0, x1, y1)), (int(side / 2 - (x1 - x0) / 2), int(side / 2 - (y1 - y0) / 2)))
    c.save(os.path.join(A, dst))
